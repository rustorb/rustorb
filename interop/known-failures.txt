# Failures that come from the existing ORBs themselves, not from the tests.
# run-matrix.sh treats these as expected, and reports a listed case that
# starts passing so that the entry can be removed.
#
# Format: [suite=SUITE] [giop=VER] CLIENT SERVER CASE REASON...
# (CLIENT / SERVER are glob patterns)
#
# A line applies to the basic suite, or with suite=SUITE to
# `run-matrix.sh --suite SUITE` runs; a line with giop=VER applies only to
# `run-matrix.sh --giop VER` runs.

# omniORBpy 4.3.4 (and 4.3.2) unmarshals a GIOP 1.2 wchar with unmarshalChar()
# instead of unmarshalWChar() on Python >= 3.3 (modules/pyMarshal.cc,
# unmarshalPyObjectWChar). Marshalling is correct, so the requests and
# replies it sends are fine, but it cannot read a wchar it receives.
omniorbpy * echo_wchar[0] omniORBpy cannot unmarshal wchar replies
omniorbpy * echo_wchar[1] omniORBpy cannot unmarshal wchar replies
omniorbpy * echo_wchar[2] omniORBpy cannot unmarshal wchar replies
* omniorbpy echo_wchar[0] omniORBpy cannot unmarshal wchar requests
* omniorbpy echo_wchar[1] omniORBpy cannot unmarshal wchar requests
* omniorbpy echo_wchar[2] omniORBpy cannot unmarshal wchar requests

# wchar / wstring do not work over GIOP 1.1 between omniORB peers. The
# omniORBpy client raises BAD_PARAM_WCharTCSNotKnown before sending. The
# omniORB C++ client sends the value in the GIOP 1.2 form anyway, and both
# servers reply with BAD_PARAM minor 0x4f4d0017 (BAD_PARAM_WCharTCSNotKnown).
# The rustorb server reads the omniORB C++ client's value as GIOP 1.1 data
# (UTF-16, as negotiated in the code set context) and echoes it, and the
# client then fails to read the reply (MARSHAL). The rustorb clients and
# servers (rustorb, rustorbpy) agree on GIOP 1.1 wchar data, so those pairs
# are not listed.
giop=1.1 omniorb* * echo_wchar[0] omniORB has no GIOP 1.1 wchar code set
giop=1.1 rustorb* omniorb* echo_wchar[0] omniORB has no GIOP 1.1 wchar code set
giop=1.1 omniorb* * echo_wchar[1] omniORB has no GIOP 1.1 wchar code set
giop=1.1 rustorb* omniorb* echo_wchar[1] omniORB has no GIOP 1.1 wchar code set
giop=1.1 omniorb* * echo_wchar[2] omniORB has no GIOP 1.1 wchar code set
giop=1.1 rustorb* omniorb* echo_wchar[2] omniORB has no GIOP 1.1 wchar code set
giop=1.1 omniorb* * echo_wstring[0] omniORB has no GIOP 1.1 wchar code set
giop=1.1 rustorb* omniorb* echo_wstring[0] omniORB has no GIOP 1.1 wchar code set
giop=1.1 omniorb* * echo_wstring[1] omniORB has no GIOP 1.1 wchar code set
giop=1.1 rustorb* omniorb* echo_wstring[1] omniORB has no GIOP 1.1 wchar code set

# JacORB 3.9's IDL to Java mapping keeps only one discriminator value for
# the default member of a union: InteropTest::Value.flag(int, boolean)
# raises BAD_OPERATION for any discriminator but 0, so the JacORB client
# cannot make Value { 99: flag }, and the JacORB server reads it as
# Value { 0: flag } and echoes that. (Found by calling the generated
# classes, not from the JacORB source.)
jacorb * echo_value[5] JacORB unions keep one default discriminator
* jacorb echo_value[5] JacORB unions keep one default discriminator
# The same generated classes on rustorb (rustorbjava-portable).
rustorbjava-portable * echo_value[5] JacORB unions keep one default discriminator
* rustorbjava-portable echo_value[5] JacORB unions keep one default discriminator

# JacORB 3.9 does not accept GIOP 1.1 Fragment messages ("Received a
# fragmented GIOP 1.1 message", then NO_IMPLEMENT). The omniORB clients
# fragment large GIOP 1.1 requests; rustorb never fragments what it sends.
giop=1.1 omniorb* jacorb echo_string[2] JacORB rejects GIOP 1.1 fragments
giop=1.1 omniorb* jacorb echo_records[large] JacORB rejects GIOP 1.1 fragments
# omniORB 4.3.4 C++ writes the state members of valuetype TypeCodes with
# visibility 0 (PRIVATE_MEMBER) where the IDL says public; omniORBpy writes
# 1 (PUBLIC_MEMBER). The omniORBpy server answers an any with the TypeCode
# of its own stubs, which the C++ client then finds not equal() to its
# own. (Seen in the captured messages.) The rustorb servers send back the
# TypeCode they received, so those pairs pass.
suite=value omniorb-cpp omniorbpy echo_any[ring] omniORB C++ valuetype TypeCodes have private members
suite=value omniorb-cpp omniorbpy echo_any[derived] omniORB C++ valuetype TypeCodes have private members

# JacORB 3.9 writes a value in an any through javax.rmi.CORBA.Util (to make
# an RMI repository id), whose implementation on the class path (JBoss's)
# needs JDK internals that Java 17 no longer has
# (sun.reflect.ReflectionFactory$GetReflectionFactoryAction). So a JacORB
# client cannot send such an any, nor a JacORB server echo one back.
suite=value jacorb * echo_any[ring] JacORB on Java 17 cannot write values in anys
suite=value jacorb * echo_any[derived] JacORB on Java 17 cannot write values in anys
suite=value jacorb * echo_any[box] JacORB on Java 17 cannot write values in anys
suite=value * jacorb echo_any[ring] JacORB on Java 17 cannot write values in anys
suite=value * jacorb echo_any[derived] JacORB on Java 17 cannot write values in anys
suite=value * jacorb echo_any[box] JacORB on Java 17 cannot write values in anys

# JacORB 3.9's valuetype Helpers: Point3Helper.type() has no concrete base
# and no truncatable modifier, and PointHelper.insert gives the any the

# finds not equal() to PointHelper.type(); a server with those Helpers
# decodes a Point3 without the members of Point. JacORB on its own ORB does
